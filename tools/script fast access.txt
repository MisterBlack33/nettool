function mtest {
    param(

    )

}

# Beispiel: mtest neue Implementierung 61 Klassen

function vtest {
    $global:LASTEXITCODE = 0
    try {
        .\tools\scripts\Verify-TestSuite.ps1 @args
        $global:LASTEXITCODE = 0
    }
    catch {
        Write-Host $_.Exception.Message -ForegroundColor Red
        $global:LASTEXITCODE = 1
    }
}

function gtest {
    param(
        [ValidateRange(1, 20)]
        [int]$Runs = 1,
        [Parameter(ValueFromRemainingArguments)]
        [string[]]$Comment
    )

    & .\tools\scripts\Run-Coverage.ps1 -Runs $Runs -Comment ($Comment -join ' ')
}
