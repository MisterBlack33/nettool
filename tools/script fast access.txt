function mtest {
    param(
        [Parameter(Position = 0, ValueFromRemainingArguments)]
        [string[]]$Comment
    )

    & .\tools\scripts\Run-Coverage.ps1 -Comment ($Comment -join ' ')
}

# Beispiel: mtest neue Implementierung 61 Klassen

function vtest {
    $global:LASTEXITCODE = 0
    try {
        .\tools\scripts\Verify-TestSuite.ps1 @args
        $global:LASTEXITCODE = 0
    }
    catch {
        Write-Host $_.Exception.Message -ForegroundColor Red
        $global:LASTEXITCODE = 1
    }
}

function gtest {
    param(
        [Parameter(ValueFromRemainingArguments)]
        [object[]]$Arguments
    )

    $runs  = 1
    $words = @($Arguments)

    # Runs nur, wenn das erste Argument explizit als String in Quotes übergeben wurde, z. B. '5'
    if ($words.Count -gt 0 -and $words[0] -is [string] -and $words[0] -match '^\d+$') {
        $n = [int]$words[0]
        if ($n -lt 1 -or $n -gt 20) {
            Write-Host "Runs muss zwischen 1 und 20 liegen." -ForegroundColor Red
            return
        }
        $runs  = $n
        $words = @($words | Select-Object -Skip 1)
    }

    & .\tools\scripts\Run-Coverage.ps1 -Runs $runs -Comment ($words -join ' ')
}
